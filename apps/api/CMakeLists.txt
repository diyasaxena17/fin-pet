cmake_minimum_required(VERSION 3.20)
project(mochi_api VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(SQLite3 REQUIRED)
find_package(Threads REQUIRED)
add_library(mochi_core src/database.cpp src/game_logic.cpp src/http_server.cpp)
target_include_directories(mochi_core PUBLIC include)
target_include_directories(mochi_core SYSTEM PUBLIC vendor)
if(TARGET SQLite3::SQLite3)
  target_link_libraries(mochi_core PUBLIC SQLite3::SQLite3 Threads::Threads)
else()
  target_link_libraries(mochi_core PUBLIC SQLite::SQLite3 Threads::Threads)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(mochi_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(mochi-api src/main.cpp)
target_link_libraries(mochi-api PRIVATE mochi_core)
target_compile_definitions(mochi-api PRIVATE MOCHI_DEFAULT_DB="${CMAKE_CURRENT_SOURCE_DIR}/data/mochi.sqlite3")
include(CTest)
if(BUILD_TESTING)
  add_executable(mochi-tests tests/core_tests.cpp)
  target_link_libraries(mochi-tests PRIVATE mochi_core)
  add_test(NAME core COMMAND mochi-tests)
endif()
