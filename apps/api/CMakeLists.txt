cmake_minimum_required(VERSION 3.20)
project(mochi_api VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(SQLite3 REQUIRED)
find_package(Threads REQUIRED)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../packages/contracts/fixtures/rules.json" MOCHI_CONTRACT_RULES)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../../packages/contracts/fixtures/rules.json")
configure_file(include/contract_rules.hpp.in generated/contract_rules.hpp @ONLY)
add_library(mochi_core src/database.cpp src/game_logic.cpp src/http_server.cpp src/finance.cpp src/engagement.cpp)
target_include_directories(mochi_core PUBLIC include "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_include_directories(mochi_core SYSTEM PUBLIC vendor)
if(TARGET SQLite3::SQLite3)
  target_link_libraries(mochi_core PUBLIC SQLite3::SQLite3 Threads::Threads)
else()
  target_link_libraries(mochi_core PUBLIC SQLite::SQLite3 Threads::Threads)
endif()
add_library(mochi_warnings INTERFACE)
target_compile_options(mochi_warnings INTERFACE $<$<CXX_COMPILER_ID:Clang,AppleClang,GNU>:-Wall;-Wextra;-Wpedantic>)
target_link_libraries(mochi_core PRIVATE mochi_warnings)
add_executable(mochi-api src/main.cpp)
target_link_libraries(mochi-api PRIVATE mochi_core mochi_warnings)
target_compile_definitions(mochi-api PRIVATE MOCHI_DEFAULT_DB="${CMAKE_CURRENT_SOURCE_DIR}/data/mochi.sqlite3")
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(mochi-tests tests/core_tests.cpp)
  target_link_libraries(mochi-tests PRIVATE mochi_core mochi_warnings)
  add_test(NAME core COMMAND mochi-tests)
  add_test(NAME http COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/http_tests.py $<TARGET_FILE:mochi-api>)
  add_test(NAME demo_startup COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/startup_tests.py $<TARGET_FILE:mochi-api>)
  add_test(NAME pi_poller COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/poller_tests.py)
  set_tests_properties(core http demo_startup pi_poller PROPERTIES TIMEOUT 45)
endif()
